### **URL Generation Process (Both Platforms):**
```javascript
// 1. Original URL
const originalUrl = "https://stars.mc/voitures/occasion/ferrari-488-gtb";

// 2. Build UTM parameters
const utmParams = {
  utm_source: "chrome_extension",
  utm_medium: "qr_code", 
  utm_campaign: "test_campaign",
  utm_term: "ferrari",
  utm_content: "test_content"

// 3. Create URL with UTM parameters
const urlWithUTM = buildUrlWithUTM(originalUrl, utmParams);
// Result: https://stars.mc/voitures/occasion/ferrari-488-gtb?utm_source=chrome_extension&utm_medium=qr_code&utm_campaign=test_campaign&utm_term=ferrari&utm_content=test_content

// 4. Create short URL for tracking

// Result: https://qr-generator-qc0kwk8ul-pierres-projects-bba7ee64.vercel.app/r/ABC123

const qrData = {
```

### **Key Functions Added to Chrome Extension:**

#### **1. URL Validation**
```javascript
isValidUrl(url) {
  try {
    new URL(url);
    return true;
  } catch {
    return false;
  }
}
```

#### **2. UTM Parameter Building**
```javascript
// Build UTM parameters (same as main app)
const utmParams = {
  utm_source: sourceInput?.value || 'chrome_extension',
  utm_medium: mediumInput?.value || 'qr_code',
  utm_campaign: campaignSelect?.value || '',
  utm_term: termInput?.value || '',
  utm_content: contentInput?.value || ''
};

// Build URL with UTM parameters (same as main app)
let urlWithUTM = originalUrl;
if (utmParams.utm_campaign || utmParams.utm_source || utmParams.utm_medium || utmParams.utm_term || utmParams.utm_content) {
  const url = new URL(originalUrl);
  Object.entries(utmParams).forEach(([key, value]) => {
    if (value && value.trim()) {
      url.searchParams.set(key, value.trim());
    }
  });
  urlWithUTM = url.toString();
}
```

#### **3. URL Shortening**
```javascript
createShortUrl(originalUrl) {
  const shortCode = this.generateShortCode();
  const baseUrl = 'https://qr-generator-qc0kwk8ul-pierres-projects-bba7ee64.vercel.app';
  return `${baseUrl}/r/${shortCode}`;
}

saveShortUrl(shortCode, originalUrl) {
  const shortUrls = JSON.parse(localStorage.getItem('shortUrls') || '{}');
  shortUrls[shortCode] = originalUrl;
  localStorage.setItem('shortUrls', JSON.stringify(shortUrls));
}
```
